      function sanitizeState(raw) {

        const s = Object.assign({}, DEFAULT_STATE, raw || {});

        s.money = Number.isFinite(s.money) && s.money >= 0 ? Math.floor(s.money) : 0;

        s.totalEarned = Number.isFinite(s.totalEarned) && s.totalEarned >= 0 ? Math.floor(s.totalEarned) : 0;

        s.tickets = Number.isFinite(s.tickets) && s.tickets >= 0 ? Math.floor(s.tickets) : 0;

        s.goldenTickets = Number.isFinite(s.goldenTickets) && s.goldenTickets >= 0 ? Math.floor(s.goldenTickets) : 0;

        s.pity = Number.isFinite(s.pity) ? Math.min(6, Math.max(0, Math.floor(s.pity))) : 0;

        s.houseTier = Number.isFinite(s.houseTier) ? Math.min(HOUSE_TIERS.length - 1, Math.max(0, Math.floor(s.houseTier))) : 0;

        s.tools = Object.assign({}, DEFAULT_STATE.tools, s.tools || {});

        s.unlockedThemes = Array.isArray(s.unlockedThemes) && s.unlockedThemes.length ? s.unlockedThemes : ['studio'];

        if (!s.unlockedThemes.includes(s.theme)) s.theme = 'studio';

        s.decors = Array.isArray(s.decors) ? s.decors : [];

        s.milestones = Array.isArray(s.milestones) ? s.milestones : [];

        s.prestige = Number.isFinite(s.prestige) && s.prestige >= 0 ? Math.floor(s.prestige) : 0;

        s.perkMult = Number.isFinite(s.perkMult) && s.perkMult >= 1 ? s.perkMult : 1.0;

        s.lottoLuck = Number.isFinite(s.lottoLuck) && s.lottoLuck >= 0 ? s.lottoLuck : 0;

        return s;

      }

      function spawnParticles(pos, colorHex, count = 8) {

        let spawned = 0;

        for (let i = 0; i < particlePool.length && spawned < count; i++) {

          const p = particlePool[i];

          if (!p.visible) {

            p.visible = true;

            p.position.copy(pos);

            p.material.color.setHex(colorHex);

            p.material.opacity = 1;

            p.userData.life = 0;

            p.userData.maxLife = 0.45 + Math.random() * 0.25;

            p.userData.vel.set((Math.random() - 0.5) * 3.5, 2 + Math.random() * 2.5, (Math.random() - 0.5) * 3.5);

            spawned++;

          }

        }

      }


      function disposeGroup(group) {

        group.traverse((obj) => {

          if (obj.geometry) obj.geometry.dispose();

          if (obj.material) {

            if (Array.isArray(obj.material)) obj.material.forEach(m => m.dispose());

            else obj.material.dispose();

          }

        });

      }

      function setPointer(e) {

        const clientX = e.touches ? e.touches[0].clientX : e.clientX;

        const clientY = e.touches ? e.touches[0].clientY : e.clientY;

        pointer.x = (clientX / window.innerWidth) * 2 - 1;

        pointer.y = -(clientY / window.innerHeight) * 2 + 1;

        return { x: clientX, y: clientY };

      }

      function collectMess(item, screenPos) {

        const idx = messItems.indexOf(item);

        if (idx === -1) return;

        messItems.splice(idx, 1);


        const tierMult = HOUSE_TIERS[state.houseTier].mult;

        const toolLv = state.tools[TOOLS.find(t => t.target === item.userData.messKind)?.id || 'hand'] || 1;

        const decorMult = state.decors.includes('plant') ? 1.15 : 1.0;

        const prestigeMult = 1 + state.prestige * 0.5;

        const reward = Math.round(item.userData.baseVal * tierMult * (1 + (toolLv - 1) * 0.35) * state.perkMult * decorMult * prestigeMult);


        state.money += reward;

        state.totalEarned += reward;

        if (item.userData.messKind === 'secret') {

          state.tickets += 1;

          showFloatingText(screenPos.x, screenPos.y, `+₩${reward.toLocaleString()} & 🎫+1`, 'text-amber-300');

        } else {

          showFloatingText(screenPos.x, screenPos.y, `+₩${reward.toLocaleString()}`);

        }


        spawnParticles(item.position, item.userData.messKind === 'secret' ? 0xfacc15 : 0x38bdf8, 10);

        roomGroup.remove(item);

        if (item.geometry) item.geometry.dispose();

        if (item.material) item.material.dispose();


        checkMilestones();

        if (messItems.length === 0) onRoomCleaned();

        else { updateHUD(); saveState(); }

      }

      const clock = new THREE.Clock();

      function animate() {

        requestAnimationFrame(animate);

        const dt = Math.min(clock.getDelta(), 0.05);

        const t = clock.getElapsedTime();

        if (droneActive && droneMesh) {

          droneMesh.position.set(Math.sin(t * 1.8) * 2.4, 2.3 + Math.sin(t * 3.5) * 0.25, Math.cos(t * 1.4) * 2.2);

          droneMesh.rotation.y += dt * 2.5;

        }

        for (let i = 0; i < particlePool.length; i++) {

          const p = particlePool[i];

          if (p.visible) {

            p.userData.life += dt;

            if (p.userData.life >= p.userData.maxLife) p.visible = false;

            else {

              p.position.addScaledVector(p.userData.vel, dt);

              p.userData.vel.y -= 6.5 * dt;

              p.material.opacity = 1 - p.userData.life / p.userData.maxLife;

            }

          }

        }

        renderer.render(scene, camera);

      }


      buildRoom();

      renderToolDock();

      updateHUD();

      animate();

    })();
